1. **TransformControls 挂载**（`setupTransformControls`）：
```ts
let transformControls: TransformControls | null = null;

const setupTransformControls = (): void => {
    const inst = engine.value;
    if (!inst || transformControls) return;
    transformControls = new TransformControls(inst.cameraRig.camera, inst.renderer.domElement);
    transformControls.name = "transform_controls";
    transformControls.setSize(0.8);
    // 拖拽开始/结束（用于提交历史栈）
    transformControls.addEventListener("dragging-changed", (event: any) => {
        if (!event.value) {
            // 拖拽结束 → 提交历史 + 同步 entity
            const attached = transformControls?.object;
            if (attached?.userData?.entityId) {
                emit("entity-transform-end", attached.userData.entityId);
            }
        }
    });
    // 拖拽中实时同步 position/size/scale 回 entity
    transformControls.addEventListener("objectChange", () => {
        const attached = transformControls?.object;
        if (!attached?.userData?.entityId) return;
        const id = attached.userData.entityId as string;
        if (transformControls!.mode === "translate") {
            emit("entity-transform", {
                id,
                position: { x: attached.position.x, y: attached.position.y, z: attached.position.z },
            });
        } else if (transformControls!.mode === "scale") {
            // scale 转 size 更新（cube: width/height/depth；cylinder/sphere: radius）
            // 由父组件根据 entity.type 决定如何映射
            emit("entity-transform", {
                id,
                position: { x: attached.position.x, y: attached.position.y, z: attached.position.z },
                size: attached.userData.originalSize // 父组件计算
                    ? undefined
                    : undefined,
            });
        }
    });
    inst.sceneManager.currentRoot.add(transformControls);
};

const attachTransformToSelected = (entityId: string | null): void => {
    if (!transformControls) return;
    if (!entityId) {
        transformControls.detach();
        return;
    }
    const inst = engine.value;
    if (!inst) return;
    // 在 currentRoot 中查找 userData.entityId === entityId 的 mesh
    let target: Object3D | null = null;
    inst.sceneManager.currentRoot.traverse((obj) => {
        if (target) return;
        if (obj.userData?.entityId === entityId) target = obj;
    });
    if (target && !(target as any).isHelper) {
        transformControls.attach(target);
    } else {
        transformControls.detach();
    }
};

const setTransformMode = (mode: "translate" | "rotate" | "scale"): void => {
    if (transformControls) transformControls.mode = mode;
};

const detachTransform = (): void => {
    transformControls?.detach();
};
```

2. **框选（Shift+拖拽）**：
```ts
let selectionBox: SelectionBox | null = null;
let selectionHelper: SelectionHelper | null = null;

const handleCanvasMouseDown = (e: MouseEvent): void => {
    if (!engine.value) return;
    // 仅 edit 模式 + Shift 触发框选
    if (!props.editMode || !e.shiftKey) return;
    const inst = engine.value;
    const canvas = canvasRef.value;
    if (!canvas) return;
    const rect = canvas.getBoundingClientRect();
    // SelectionBox 使用 NDC 坐标
    selectionBox = new SelectionBox(inst.cameraRig.camera, inst.sceneManager.currentRoot);
    selectionHelper = new SelectionHelper(selectionBox, inst.renderer, "dt-box-select-helper");
    selectionHelper.onMouseDown(e);
    // 收集可选 mesh（排除锁定/隐藏/辅助物体）
    selectionBox.startPoint.set(
        ((e.clientX - rect.left) / rect.width) * 2 - 1,
        -((e.clientY - rect.top) / rect.height) * 2 + 1,
        0.5,
    );
    selectionBox.endPoint.copy(selectionBox.startPoint);
    document.addEventListener("mousemove", handleBoxMouseMove);
    document.addEventListener("mouseup", handleBoxMouseUp);
};

const handleBoxMouseMove = (e: MouseEvent): void => {
    if (!selectionBox || !selectionHelper || !canvasRef.value) return;
    const rect = canvasRef.value.getBoundingClientRect();
    selectionBox.endPoint.set(
        ((e.clientX - rect.left) / rect.width) * 2 - 1,
        -((e.clientY - rect.top) / rect.height) * 2 + 1,
        0.5,
    );
    selectionBox.select();
    selectionHelper.onMouseMove(e);
};

const handleBoxMouseUp = (e: MouseEvent): void => {
    document.removeEventListener("mousemove", handleBoxMouseMove);
    document.removeEventListener("mouseup", handleBoxMouseUp);
    if (!selectionBox) return;
    const ids = selectionBox.collection
        .filter((obj) => obj.userData?.entityId && !props.lockedIds.includes(obj.userData.entityId))
        .map((obj) => obj.userData.entityId as string);
    const uniqueIds = [...new Set(ids)];
    if (uniqueIds.length > 0) {
        emit("box-select", uniqueIds);
    }
    selectionHelper?.onMouseUp(e);
    selectionBox = null;
    selectionHelper = null;
};
```

```ts
    ndc.y = -((e.clientY - rect.top) / rect.height) * 2 + 1;
    raycaster.setFromCamera(ndc, inst.cameraRig.camera);
    const intersects = raycaster.intersectObjects(inst.sceneManager.currentRoot.children, true);
    const hit = intersects.find(
        (i) => !(i.object instanceof GridHelper) && !(i.object instanceof AxesHelper) && !(i.object instanceof Group) && !(i.object as any).isTransformControls,
    );
    const entityId = hit?.object?.userData?.entityId as string | null;
    if (entityId) {
        // 选中右键点击的实体
        sceneApi.selectNode(hit!.object.name || hit!.object.uuid);
        emit("select-entity", entityId);
    }
    emit("context-menu", { x: e.clientX, y: e.clientY, entityId: entityId ?? null });
};
```

```vue
import DtLegendPanel from "./panels/DtLegendPanel.vue";
import DtContextMenu, { type DtContextMenuItem } from "./panels/DtContextMenu.vue";
import DtEntityDialog from "./dialogs/DtEntityDialog.vue";
import DtCustomComponentDialog from "./dialogs/DtCustomComponentDialog.vue";

import { useDigitalTwinCamera } from "./composables/useDigitalTwinCamera";
import { useDigitalTwinTelemetry } from "./composables/useDigitalTwinTelemetry";
import type { DtEntity, DtPoint3D, DtScene, DtSize3D, DtViewPreset } from "./types";

const props = defineProps<{ scene?: DtScene; autoSim?: boolean; }>();
const emit = defineEmits<{ (e: "select-entity", id: string | null): void; (e: "ready"): void; }>();

const designer = useDigitalTwinDesigner(props.scene ? { ...props.scene } : undefined);
const camera = useDigitalTwinCamera();
const telemetry = useDigitalTwinTelemetry();
const viewportRef = ref<InstanceType<typeof DtSceneViewport> | null>(null);

// ─── 弹窗状态 ───
const dialogVisible = ref(false);
const editingEntityId = ref<string | null>(null);
const editingSourceEntity = ref<DtEntity | null>(null);
const customDialogVisible = ref(false);

// ─── 预览模式 ───
const isPreview = ref(false);
const transformMode = ref<"translate" | "rotate" | "scale">("translate");

// ─── 右键菜单状态 ───
const ctxMenuVisible = ref(false);
const ctxMenuPos = ref({ x: 0, y: 0 });
const ctxMenuContext = ref<"canvas" | "entity">("canvas");

// ─── 多选（框选）状态 ───
const selectedIds = ref<string[]>([]);

// ─── 文件导入 input ───
const fileInputRef = ref<HTMLInputElement | null>(null);

// ─── 计算：锁定的实体 ID 列表（供视口跳过拾取） ───
const lockedIds = computed(() =>
    designer.state.scene.entities.filter((e) => e.locked).map((e) => e.id),
);

// ─── 视口 editMode：非预览模式即编辑 ───
const viewportEditMode = computed(() => !isPreview.value);

// ─── 工具栏事件 ───
const handleNew = (): void => {
    ElMessageBox.confirm(i18n("digitaltwin.message.sceneCleared" as any), i18n("digitaltwin.menu.new" as any), { type: "warning" })
        .then(() => { designer.newScene(); ElMessage.success(i18n("digitaltwin.message.sceneCleared" as any)); })
        .catch(() => {});
};
const handleImport = (): void => { fileInputRef.value?.click(); };
const handleImportFile = (e: Event): void => {
    const target = e.target as HTMLInputElement;
    const file = target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = () => {
        try { designer.importSceneJson(String(reader.result)); }
        catch (err) { ElMessage.error(i18n("digitaltwin.message.invalidSceneFile" as any)); }
    };
    reader.readAsText(file);
    target.value = "";
};
const handleExport = (): void => {
    const json = designer.exportSceneJson();
    const blob = new Blob([json], { type: "application/json" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = `digital_twin_scene_${Date.now()}.json`;
    document.body.appendChild(a);
    a.click();
    document.body.removeChild(a);
    URL.revokeObjectURL(url);
    ElMessage.success(i18n("digitaltwin.message.sceneExported" as any));
};
const handleSave = (): void => {
    const scene = designer.saveScene();
    // 这里仅做内存保存 + 提示；真实项目可对接后端 API
    ElMessage.success(i18n("digitaltwin.message.sceneSaved" as any));
    // 同时下载一份 JSON 副本
    handleExport();
};
const handleUndo = (): void => designer.undo();
const handleRedo = (): void => designer.redo();
const handlePreview = (): void => { isPreview.value = !isPreview.value; };
const handleCustomComponent = (): void => { customDialogVisible.value = true; };
const handleTransformModeChange = (mode: "translate" | "rotate" | "scale"): void => {
    transformMode.value = mode;
    viewportRef.value?.setTransformMode(mode);
};

// ─── 视口事件 ───
const handleEntityTransform = (payload: { id: string; position: DtPoint3D; size?: DtSize3D }): void => {
    // 实时更新（不入历史栈，拖拽结束时入栈）
    const patch: Partial<DtEntity> = { position: payload.position };
    if (payload.size) patch.size = payload.size;
    designer.updateEntity(payload.id, patch);
};
const handleEntityTransformEnd = (id: string): void => {
    designer.commitUpdate();
};
const handleBoxSelect = (ids: string[]): void => {
    selectedIds.value = ids;
    designer.selectEntity(ids.length > 0 ? ids[ids.length - 1] : null);
    ElMessage.info(i18n("digitaltwin.message.boxSelected" as any).replace("{0}", String(ids.length)));
};
const handleContextMenu = (payload: { x: number; y: number; entityId: string | null }): void => {
    ctxMenuPos.value = { x: payload.x, y: payload.y };
    ctxMenuContext.value = payload.entityId ? "entity" : "canvas";
    ctxMenuVisible.value = true;
};

// ─── 右键菜单 action 路由 ───
const handleCtxAction = (action: string): void => {
    switch (action) {
        case "new": handleNew(); break;
        case "undo": handleUndo(); break;
        case "redo": handleRedo(); break;
        case "preview": handlePreview(); break;
        case "import": handleImport(); break;
        case "export": handleExport(); break;
        case "customComponent": handleCustomComponent(); break;
        case "save": handleSave(); break;
        case "duplicate":
            if (designer.state.selectedEntityId) designer.duplicateEntity(designer.state.selectedEntityId);
            break;
        case "bringFront":
            if (designer.state.selectedEntityId) {
                // 重复 moveEntityUp 直至置顶
                const id = designer.state.selectedEntityId;
                const arr = designer.state.scene.entities;
                const idx = arr.findIndex((e) => e.id === id);
                for (let i = idx; i > 0; i--) designer.moveEntityUp(id);
            }
            break;
        case "sendBack":
            if (designer.state.selectedEntityId) {
                const id = designer.state.selectedEntityId;
                const arr = designer.state.scene.entities;
                const idx = arr.findIndex((e) => e.id === id);
                for (let i = idx; i < arr.length - 1; i++) designer.moveEntityDown(id);
            }
            break;
        case "delete":
            if (designer.state.selectedEntityId) {
                designer.removeEntity(designer.state.selectedEntityId);
                ElMessage.success(i18n("digitaltwin.message.deleted"));
            }
            break;
    }
};

// ─── 选中实体变化 → 视口 attachTransformToSelected ───
watch(
    () => designer.state.selectedEntityId,
    (id) => {
        if (viewportEditMode.value) {
            viewportRef.value?.attachTransformToSelected(id);
        }
        emit("select-entity", id);
    },
);

// ─── 预览模式切换 ───
watch(isPreview, (preview) => {
    if (preview) {
        viewportRef.value?.detachTransform();
    } else {
        // 退出预览：重新 attach 到当前选中
        viewportRef.value?.attachTransformToSelected(designer.state.selectedEntityId);
    }
});

// ─── 全局键盘快捷键 ───
const handleKeydown = (e: KeyboardEvent): void => {
    // 在 input/textarea 中不触发
    const target = e.target as HTMLElement;
    if (target.tagName === "INPUT" || target.tagName === "TEXTAREA" || target.isContentEditable) return;
    const ctrl = e.ctrlKey || e.metaKey;
    if (ctrl && e.key === "z" && !e.shiftKey) { e.preventDefault(); handleUndo(); return; }
    if (ctrl && (e.key === "y" || (e.key === "z" && e.shiftKey))) { e.preventDefault(); handleRedo(); return; }
    if (ctrl && e.key === "s") { e.preventDefault(); handleSave(); return; }
    if (ctrl && e.key === "d" && designer.state.selectedEntityId) {
        e.preventDefault();
        designer.duplicateEntity(designer.state.selectedEntityId);
        return;
    }
    if ((e.key === "Delete" || e.key === "Backspace") && designer.state.selectedEntityId) {
        e.preventDefault();
        designer.removeEntity(designer.state.selectedEntityId);
        ElMessage.success(i18n("digitaltwin.message.deleted"));
        return;
    }
};
onMounted(() => { document.addEventListener("keydown", handleKeydown); });
onBeforeUnmount(() => {
    document.removeEventListener("keydown", handleKeydown);
    telemetry.stopSim();
});

// ─── 其余已有逻辑（addEntity/editEntity/dialogSubmit/selectEntity/viewChange/toggleSim）保持不变 ───
// （略，直接复用现有实现）

// 自动开启模拟（保持不变）
watch(() => props.autoSim, (auto) => {
    if (auto && !telemetry.simRunning.value) {
        telemetry.startSim(() => designer.state.scene.entities);
        designer.setSimRunning(true);
    }
}, { immediate: true });

const simRunning = computed(() => telemetry.simRunning.value);

// 视角切换（保持不变，但增加同步 transformMode）
const handleViewChange = (preset: DtViewPreset): void => {
    camera.applyViewPreset(preset);
    viewportRef.value?.applyPreset(preset as any).catch(() => void 0);
};

const handleToggleSim = (): void => {
    if (telemetry.simRunning.value) {
        telemetry.stopSim(); designer.setSimRunning(false);
        ElMessage.info(i18n("digitaltwin.message.simStopped"));
    } else {
        telemetry.startSim(() => designer.state.scene.entities);
        designer.setSimRunning(true);
        ElMessage.success(i18n("digitaltwin.message.simStarted"));
    }
};

const handleAddEntity = (): void => {
    editingEntityId.value = null; editingSourceEntity.value = null; dialogVisible.value = true;
};
const handleEditEntity = (id: string): void => {
    const entity = designer.state.scene.entities.find((e) => e.id === id);
    if (!entity) return;
    editingEntityId.value = id; editingSourceEntity.value = entity; dialogVisible.value = true;
};
const handleDialogSubmit = (entity: DtEntity, isNew: boolean): void => {
    if (isNew) { designer.addEntity(entity); ElMessage.success(i18n("digitaltwin.message.added")); }
    else { designer.updateEntity(entity.id, entity); designer.commitUpdate(); ElMessage.success(i18n("digitaltwin.message.updated")); }
};
const handleSelectEntity = (id: string | null): void => {
    if (id !== designer.state.selectedEntityId) designer.selectEntity(id);
    emit("select-entity", id);
};
const handleSelectNode = (path: string | null): void => {
    if (!path) return;
    const viewport = viewportRef.value as any;
    const engine = viewport?.engine?.value;
    if (!engine) return;
    const node = engine.sceneManager.findByPath(path);
    const entityId = node?.object3D?.userData?.entityId as string | undefined;
    if (entityId && entityId !== designer.state.selectedEntityId) {
        designer.selectEntity(entityId);
        emit("select-entity", entityId);
    }
};
const handleViewportReady = (): void => { emit("ready"); };
const handleViewportError = (err: Error): void => { ElMessage.error(err.message || String(err)); };

// watch scene（保持不变）
watch(() => props.scene, (s) => { if (s) designer.loadScene(s); }, { deep: false });

defineExpose({ designer, camera, telemetry, viewportRef });
</script>
```

## 四、假设与决策

| 假设/决策 | 说明 |
|---|---|
| el-splitter 可用 | Element Plus 2.14.2 ≥ 2.6.0（el-splitter 引入版本），直接采用 |
| TransformControls 与 SelectionBox/SelectionHelper 三件套 import 路径 | `three/examples/jsm/controls/TransformControls.js` + `three/examples/jsm/interactive/SelectionBox.js` + `three/examples/jsm/interactive/SelectionHelper.js`（已确认存在） |
| 无 OrbitControls 需停用 | 当前视口仅预设视角，无实时轨道控制器；TransformControls 的 `dragging-changed` 仅用于提交历史栈。后续若接入 OrbitControls 再加 `inst.cameraRig.orbitControls.enabled = !event.value` |
| Scale Gizmo → size 映射 | 由父组件（DigitalTwinDesigner）根据 entity.type 决定如何把 scale 转 size（cube: width/height/depth 直接乘 scale；cylinder/sphere: radius 乘 scale.x）。本批次先实现 translate（用户最核心诉求），rotate/scale 在 transformMode 切换后由 Three.js 内部处理几何变换，回写 size 留到下批次精细化 |
| 自定义组件持久化 | 使用 localStorage（演示场景足够）；真实项目可改为后端 API。运行时 i18n 通过 `window.__dtCustomI18n` 注入，避免新增 key 后显示字面量 |
| 框选触发键 | Shift+左键拖拽（行业标准，与 Blender/Unity 一致） |
| 右键菜单触发 | canvas `@contextmenu.prevent` 阻止浏览器默认菜单，派发坐标 + entityId 给 DtContextMenu |
| 全局快捷键作用域 | INPUT/TEXTAREA/contentEditable 中不触发，避免与表单输入冲突 |
| 保存语义 | 当前为内存保存 + 自动下载 JSON 副本（双保险）；真实项目对接后端时仅需替换 handleSave 实现 |
| 预览模式 | 切换 isPreview：隐藏左右 el-splitter-panel（条件渲染）、隐藏浮动图例、视口切 view 模式、detach Gizmo |
| 移除 `as any` | B12 完成后，所有 `i18n("...key..." as any)` 改为 `i18n("...key...")`；自定义组件的动态 key（`digitaltwin.custom.${type}`）保留 `as any` 因为是运行时生成 |
